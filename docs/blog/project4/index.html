import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans

# Load the Palmer Penguins dataset
df = pd.read_csv('/Users/danielwang/Desktop/UCSD Spring/MGTA495 Marketing Analytics/Website/blog/project4/data/palmer_penguins.csv')
data = df[['bill_length_mm', 'flipper_length_mm']].dropna().values

# Scratch implementation of K-Means
def kmeans_scratch(data, k, max_iters=100):
    np.random.seed(42)
    # Initialize centroids randomly from the data points
    centroids = data[np.random.choice(data.shape[0], k, replace=False)]
    history = [centroids.copy()]
    
    for i in range(max_iters):
        # Compute distances and assign labels
        distances = np.linalg.norm(data[:, np.newaxis] - centroids, axis=2)
        labels = np.argmin(distances, axis=1)
        
        # Update centroids
        new_centroids = np.array([data[labels == j].mean(axis=0) for j in range(k)])
        history.append(new_centroids.copy())
        
        # Check for convergence
        if np.allclose(centroids, new_centroids):
            break
        
        centroids = new_centroids
    
    return labels, centroids, history

# Run scratch K-Means (k=3)
labels_scratch, centroids_scratch, history = kmeans_scratch(data, k=3)

# Plot centroids progression
for i, cent in enumerate(history):
    plt.figure(figsize=(6, 4))
    title = 'Initial Centroids' if i == 0 else f'Iteration {i}'
    plt.title(title)
    plt.scatter(data[:, 0], data[:, 1], s=10, alpha=0.6)
    plt.scatter(cent[:, 0], cent[:, 1], marker='X', s=100, color='red')
    plt.xlabel('Bill Length (mm)')
    plt.ylabel('Flipper Length (mm)')
    plt.tight_layout()
    plt.show()

from sklearn.metrics import silhouette_score

# Prepare metrics
ks = range(2, 8)
inertias = []
silhouettes = []

for k in ks:
    km = KMeans(n_clusters=k, random_state=42).fit(data)

    # silhouette requires at least 2 clusters and less than n_samples clusters
    sil_score = silhouette_score(data, km.labels_)

from collections import Counter
from sklearn.neighbors import KNeighborsClassifier
from sklearn.metrics import accuracy_score

# --- Generate training data (seed=42) ---
np.random.seed(42)
n_train = 100
x1_train = np.random.uniform(-3, 3, size=n_train)
x2_train = np.random.uniform(-3, 3, size=n_train)
boundary_train = np.sin(4 * x1_train) + x1_train
y_train = np.where(x2_train > boundary_train, 1, 0)
X_train = np.column_stack((x1_train, x2_train))

# --- Generate test data (seed=24) ---
np.random.seed(24)
n_test = 100
x1_test = np.random.uniform(-3, 3, size=n_test)
x2_test = np.random.uniform(-3, 3, size=n_test)
boundary_test = np.sin(4 * x1_test) + x1_test
y_test = np.where(x2_test > boundary_test, 1, 0)
X_test = np.column_stack((x1_test, x2_test))

# --- Scratch KNN implementation ---
def knn_predict(X_train, y_train, X_query, k=3):
    preds = []
    for q in X_query:
        dists = np.linalg.norm(X_train - q, axis=1)
        neighbors = np.argsort(dists)[:k]
        vote = Counter(y_train[neighbors]).most_common(1)[0][0]
        preds.append(vote)
    return np.array(preds)

# Run hand-rolled KNN
k = 3
y_pred_scratch = knn_predict(X_train, y_train, X_test, k=k)
acc_scratch = accuracy_score(y_test, y_pred_scratch)

# --- scikit-learn KNN for verification ---
knn = KNeighborsClassifier(n_neighbors=k)
knn.fit(X_train, y_train)
y_pred_sklearn = knn.predict(X_test)
acc_sklearn = accuracy_score(y_test, y_pred_sklearn)

# --- Results ---
results = pd.DataFrame({
    'Method': ['Scratch KNN', 'scikit-learn KNN'],
    'Accuracy': [acc_scratch, acc_sklearn]
})
print(results)
